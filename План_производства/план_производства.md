# План производства

**Задача** 
- Создать дашборд или аналитическую таблицу плана производства оборудования

**Таблица**
- Заказ на производства/срок отгрузки(дата)
- Заказчик
- Обозначения
- Количество. шт
- Цена. руб
- Объем. руб
- Плановая рентабильность
- Материал - Карточка материала (Какая инфа нужна - сталь 45, ф120, расход)
- Сроки производства (с ганта) 
    - Начало
    - Окончание
- Сроки подготовки
    - Технолог УП (дата)
    - Оснастка (дата)
    - Материал (дата)

)t2

SELECT
    b.DocEntry AS [Номер заказа],

    b.BELPOS_ID AS [Поз ПЗ],
    b.KND_ID AS [Код заказчика],
    b.KNDNAME AS [Имя заказчика],
    b.ItemCode AS [Код детали],
    b.ItemName AS [Описание детали],

    b.LFGDAT AS [Сроки отгрузки (LFGDAT)],
    b.GanttOrderEnd AS [Сроки по Ганту (ENDZEIT)],

        WHEN CAST(b.GanttOrderEnd AS DATE) <= CAST(b.LFGDAT AS DATE) THEN N'Успеваем'
        ELSE N'Не успеваем'
    END AS [Проверка Ганта],

    fs.FactStartDate AS [Факт начала обработки],

    b.RequiredQty AS [Требуется мат-ла],
    m.MaterialIssuedQty AS [Подано мат-ла],
    CASE WHEN ISNULL(m.MaterialIssuedQty, 0) >= ISNULL(b.RequiredQty, 0) THEN '+' ELSE '-' END AS [Материал подан?],
    m.MaterialDescription AS [Описание заготовки],

    CASE WHEN up.UP_Value IN (N'Программное', N'Факт', N'Другой станок') THEN 'J' ELSE 'N' END AS [УП Готова? (J/N)],

LEFT JOIN #UPStatus up ON b.BELNR_ID = up.BELNR_ID AND b.BELPOS_ID = up.BELPOS_ID
LEFT JOIN #ToolAgg ta ON b.BELNR_ID = ta.BELNR_ID AND b.BELPOS_ID = ta.BELPOS_ID
LEFT JOIN dbo.RDR1 s ON b.DocEntry = s.DocEntry AND b.AUFTRAGPOS = s.LineNum
LEFT JOIN #Finance fin ON b.BELNR_ID = fin.BELNR_ID AND b.BELPOS_ID = fin.BELPOS_ID
CROSS JOIN #DateRange d

WHERE
(
    (b.LFGDAT >= d.FilterStart AND b.LFGDAT <= d.FilterEnd)
    OR (b.GanttOrderEnd >= d.FilterStart AND b.GanttOrderEnd <= d.FilterEnd)
    OR (fs.FactStartDate >= d.FilterStart AND fs.FactStartDate <= d.FilterEnd)
    OR (
        ISNULL(fs.FactStartDate, b.LFGDAT) <= d.FilterEnd
        AND ISNULL(b.GanttOrderEnd, b.LFGDAT) >= d.FilterStart
    )
)
ORDER BY b.DocEntry, b.BELNR_ID, b.BELPOS_ID;

        MergedIssuedQty = NULLIF(ISNULL(m.MaterialIssuedQty, 0) + ISNULL(c_m.MaterialIssuedQty, 0), 0),
        MergedMatDesc = STUFF(
            COALESCE(', ' + m.MaterialDescription, '') + 
            COALESCE(', ' + c_m.MaterialDescription, ''), 
        1, 2, ''),
        MergedUPValue = COALESCE(up.UP_Value, c_up.UP_Value),
        MergedUPDate = COALESCE(up.UP_Date, c_up.UP_Date),
        MergedToolList = STUFF(
            COALESCE(', ' + ta.ToolingList, '') + 
            COALESCE(', ' + c_ta.ToolingList, ''), 
        1, 2, ''),
        MergedToolReady = CASE 
            WHEN ta.ToolingReady = 'N' OR c_ta.ToolingReady = 'N' THEN 'N'
            WHEN ta.ToolingReady = 'J' OR c_ta.ToolingReady = 'J' THEN 'J'
            ELSE NULL END,
        MergedToolDate = CASE 
            WHEN ta.ToolingReadyDate IS NULL THEN c_ta.ToolingReadyDate
            WHEN c_ta.ToolingReadyDate IS NULL THEN ta.ToolingReadyDate
            WHEN ta.ToolingReadyDate > c_ta.ToolingReadyDate THEN ta.ToolingReadyDate 
            ELSE c_ta.ToolingReadyDate END
) merged

WHERE
(
    (b.LFGDAT >= d.FilterStart AND b.LFGDAT <= d.FilterEnd)
    OR (b.GanttOrderEnd >= d.FilterStart AND b.GanttOrderEnd <= d.FilterEnd)
    OR (merged.MergedFactStart >= d.FilterStart AND merged.MergedFactStart <= d.FilterEnd)
    OR (
        ISNULL(merged.MergedFactStart, b.LFGDAT) <= d.FilterEnd
        AND ISNULL(b.GanttOrderEnd, b.LFGDAT) >= d.FilterStart
    )
)
ORDER BY b.DocEntry, b.BELNR_ID, b.BELPOS_ID;

LEFT JOIN #Material c_m ON b.BELNR_ID = c_m.BELNR_ID AND cm.ChildPos = c_m.BELPOS_ID
LEFT JOIN #UPStatus c_up ON b.BELNR_ID = c_up.BELNR_ID AND cm.ChildPos = c_up.BELPOS_ID
LEFT JOIN #ToolAgg c_ta ON b.BELNR_ID = c_ta.BELNR_ID AND cm.ChildPos = c_ta.BELPOS_ID

-- Джойним продажи и финансы (финансы берем по текущей строке, т.к. они считаются на 21)
LEFT JOIN #SalesData s ON b.BELNR_ID = s.BELNR_ID AND b.BELPOS_ID = s.BELPOS_ID
LEFT JOIN #Finance fin ON b.BELNR_ID = fin.BELNR_ID AND b.BELPOS_ID = fin.BELPOS_ID

CROSS JOIN #DateRange d

-- УМНОЕ ОБЪЕДИНЕНИЕ ДАННЫХ В СТРОКУ ОТЧЕТА
OUTER APPLY (
    SELECT 
        MergedFactStart = CASE 
            WHEN fs.FactStartDate IS NULL THEN c_fs.FactStartDate
            WHEN c_fs.FactStartDate IS NULL THEN fs.FactStartDate
            WHEN fs.FactStartDate < c_fs.FactStartDate THEN fs.FactStartDate
            ELSE c_fs.FactStartDate
        END,
        MergedIssuedQty = NULLIF(ISNULL(m.MaterialIssuedQty, 0) + ISNULL(c_m.MaterialIssuedQty, 0), 0),
        MergedMatDesc = STUFF(
            COALESCE(', ' + m.MaterialDescription, '') + 
            COALESCE(', ' + c_m.MaterialDescription, ''), 
        1, 2, ''),
        MergedUPValue = COALESCE(up.UP_Value, c_up.UP_Value),
        MergedUPDate = COALESCE(up.UP_Date, c_up.UP_Date),
        MergedToolList = STUFF(
            COALESCE(', ' + ta.ToolingList, '') + 
            COALESCE(', ' + c_ta.ToolingList, ''), 
        1, 2, ''),
        MergedToolReady = CASE 
            WHEN ta.ToolingReady = 'N' OR c_ta.ToolingReady = 'N' THEN 'N'
            WHEN ta.ToolingReady = 'J' OR c_ta.ToolingReady = 'J' THEN 'J'
            ELSE NULL END,
        MergedToolDate = CASE 
            WHEN ta.ToolingReadyDate IS NULL THEN c_ta.ToolingReadyDate
            WHEN c_ta.ToolingReadyDate IS NULL THEN ta.ToolingReadyDate
            WHEN ta.ToolingReadyDate > c_ta.ToolingReadyDate THEN ta.ToolingReadyDate 
            ELSE c_ta.ToolingReadyDate END
) merged

WHERE
(
    (b.LFGDAT >= d.FilterStart AND b.LFGDAT <= d.FilterEnd)
    OR (b.GanttOrderEnd >= d.FilterStart AND b.GanttOrderEnd <= d.FilterEnd)
    OR (merged.MergedFactStart >= d.FilterStart AND merged.MergedFactStart <= d.FilterEnd)
    OR (
        ISNULL(merged.MergedFactStart, b.LFGDAT) <= d.FilterEnd
        AND ISNULL(b.GanttOrderEnd, b.LFGDAT) >= d.FilterStart
    )
)
ORDER BY b.DocEntry, b.BELNR_ID, b.BELPOS_ID;

        WHEN md.IsReady = 1 THEN md.UP_Value
        -- 3. Если нигде не готова -> берем что есть (приоритет ПЗ)
        ELSE COALESCE(wo.UP_Value, md.UP_Value)
    END AS UP_Value,

    CASE 
        WHEN wo.IsReady = 1 THEN COALESCE(wo.UP_Date, md.UP_Date)
        WHEN md.IsReady = 1 THEN md.UP_Date
        ELSE COALESCE(wo.UP_Date, md.UP_Date)
    END AS UP_Date
INTO #UPStatus
FROM #UPLookup ul

            WHEN fs.FactStartDate < c_fs.FactStartDate THEN fs.FactStartDate

        MergedUPValue = CASE

            WHEN up.UP_Value IN (N'Программное', N'Факт', N'Другой станок') THEN up.UP_Value

            WHEN ta.ToolingReady = 'J' OR c_ta.ToolingReady = 'J' THEN 'J'
            ELSE NULL END,
        MergedToolDate = CASE

            WHEN c_ta.ToolingReadyDate IS NULL THEN ta.ToolingReadyDate
            WHEN ta.ToolingReadyDate > c_ta.ToolingReadyDate THEN ta.ToolingReadyDate 
            ELSE c_ta.ToolingReadyDate END,
        MergedPricePerMin = COALESCE(c_fin.PricePerMin, fin.PricePerMin)
) merged

    b.ItemCode NOT LIKE '21%'
    AND (
        (b.LFGDAT >= d.FilterStart AND b.LFGDAT <= d.FilterEnd)
        OR (b.GanttOrderEnd >= d.FilterStart AND b.GanttOrderEnd <= d.FilterEnd)
        OR (merged.MergedFactStart >= d.FilterStart AND merged.MergedFactStart <= d.FilterEnd)
        OR (
            ISNULL(merged.MergedFactStart, b.LFGDAT) <= d.FilterEnd
            AND ISNULL(b.GanttOrderEnd, b.LFGDAT) >= d.FilterStart
        )
    )
ORDER BY b.DocEntry, b.BELNR_ID, b.BELPOS_ID;

        LTRIM(RTRIM(CAST(CAST(p.ItemCode AS NVARCHAR(MAX)) AS NVARCHAR(100)))) AS WKZ_ID,
        LTRIM(RTRIM(CAST(CAST(p.ABGKZ AS NVARCHAR(MAX)) AS NVARCHAR(10)))) AS ABGKZ,
        COALESCE(p.ABGKZ_DATE, p.ENDZEIT, p.AndTStamp, p.ErfTStamp) AS ReadyDate,
        ROW_NUMBER() OVER(PARTITION BY LTRIM(RTRIM(CAST(CAST(p.ItemCode AS NVARCHAR(MAX)) AS NVARCHAR(100)))) 
                          ORDER BY COALESCE(p.ErfTStamp, p.AndTStamp, '1900-01-01') DESC, p.BELNR_ID DESC) as rn
    FROM dbo.BEAS_FTPOS p
    WHERE LTRIM(RTRIM(CAST(CAST(p.ItemCode AS NVARCHAR(MAX)) AS NVARCHAR(100)))) LIKE '5000%'
) t
WHERE rn = 1;

CREATE CLUSTERED INDEX CIX_LatestTool ON #LatestToolOrders (WKZ_ID);

SELECT 
    rt.BELNR_ID, 
    rt.BELPOS_ID,
    CASE WHEN SUM(CASE WHEN ISNULL(lto.ABGKZ, 'N') <> 'J' THEN 1 ELSE 0 END) = 0 THEN 'J' ELSE 'N' END AS ToolingReady,
    MAX(CASE WHEN lto.ABGKZ = 'J' THEN lto.ReadyDate ELSE NULL END) AS ToolingReadyDate,
    STUFF((
        SELECT ', ' + rt2.WKZ_ID + ' (' + CASE WHEN lto2.ABGKZ = 'J' THEN '+' ELSE '-' END + ')'
        FROM #ReqTools rt2
        LEFT JOIN #LatestToolOrders lto2 ON rt2.WKZ_ID = lto2.WKZ_ID
        WHERE rt2.BELNR_ID = rt.BELNR_ID AND rt2.BELPOS_ID = rt.BELPOS_ID
        ORDER BY rt2.WKZ_ID
        FOR XML PATH(''), TYPE).value('.', 'NVARCHAR(MAX)'), 1, 2, '') AS ToolingList
INTO #ToolAgg
FROM #ReqTools rt
LEFT JOIN #LatestToolOrders lto ON rt.WKZ_ID = lto.WKZ_ID
GROUP BY rt.BELNR_ID, rt.BELPOS_ID;

CREATE CLUSTERED INDEX CIX_ToolAgg ON #ToolAgg (BELNR_ID, BELPOS_ID);

-- 8. ДАННЫЕ ПРОДАЖ (RDR1)
-------------------------------------------------------------------------------
SELECT 
    b.BELNR_ID,
    b.BELPOS_ID,
    s.DocEntry,
    s.LineNum,
    s.SalesQty,
    s.SalesPrice,
    s.Currency,
    s.OpenSumFC
INTO #SalesData
FROM #AllPositions b
OUTER APPLY (
    SELECT TOP 1 
        r.DocEntry,
        r.LineNum,
        r.Quantity AS SalesQty,
        r.Price AS SalesPrice,
        r.Currency,
        r.OpenSumFC
    FROM dbo.RDR1 r
    WHERE r.DocEntry = b.DocEntry
      AND (
          (b.AUFTRAGPOS IS NOT NULL AND r.LineNum = b.AUFTRAGPOS AND RIGHT(r.ItemCode, 7) = RIGHT(b.ItemCode, 7))
          OR (RIGHT(r.ItemCode, 7) = RIGHT(b.ItemCode, 7))
      )
    ORDER BY 
        CASE WHEN b.AUFTRAGPOS IS NOT NULL AND r.LineNum = b.AUFTRAGPOS AND RIGHT(r.ItemCode, 7) = RIGHT(b.ItemCode, 7) THEN 1 ELSE 2 END,
        r.LineNum ASC
) s
WHERE b.DocEntry IS NOT NULL;

CREATE CLUSTERED INDEX CIX_SalesData ON #SalesData (BELNR_ID, BELPOS_ID);

-------------------------------------------------------------------------------
-- 9. ФИНАНСЫ (РЕНТАБЕЛЬНОСТЬ ИЗ ВЬЮХИ GC_FACT_FINANCIAL_REPORT)

SELECT 
    f.BELNR_ID, 
    f.BELPOS_ID, 
    AVG(f.Price_for_1_min) AS PricePerMin
INTO #Finance
FROM dbo.GC_FACT_FINANCIAL_REPORT f
INNER JOIN #AllPositions b ON f.BELNR_ID = b.BELNR_ID AND f.BELPOS_ID = b.BELPOS_ID
GROUP BY f.BELNR_ID, f.BELPOS_ID;

CREATE CLUSTERED INDEX CIX_Finance ON #Finance (BELNR_ID, BELPOS_ID);

-- 10. ИТОГОВАЯ ВЫБОРКА С НАСЛЕДОВАНИЕМ В РОДИТЕЛЯ (43...)
-------------------------------------------------------------------------------
SELECT
    b.DocEntry AS [Номер заказа],
    b.BELNR_ID AS [ПЗ ID],
    b.BELPOS_ID AS [Поз ПЗ],
    b.KND_ID AS [Код заказчика],
    b.KNDNAME AS [Имя заказчика],
    b.ItemCode AS [Код детали],
    b.ItemName AS [Описание детали],
    b.PosStatus AS [Статус ПЗ (J/N)],

    b.LFGDAT AS [Сроки отгрузки (LFGDAT)],
    b.GanttOrderEnd AS [Сроки по Ганту (ENDZEIT)],

    CASE
        WHEN b.GanttOrderEnd IS NULL OR b.LFGDAT IS NULL THEN NULL
        WHEN CAST(b.GanttOrderEnd AS DATE) <= CAST(b.LFGDAT AS DATE) THEN N'Успеваем'
        ELSE N'Не успеваем'
    END AS [Проверка Ганта],

    merged.MergedFactStart AS [Факт начала обработки],

    b.RequiredQty AS [Требуется мат-ла],
    merged.MergedIssuedQty AS [Подано мат-ла],
    CASE WHEN merged.MergedIssuedQty >= ISNULL(b.RequiredQty, 0) THEN '+' ELSE '-' END AS [Материал подан?],
    merged.MergedMatDesc AS [Описание заготовки],

    -- УП
    CASE WHEN merged.MergedUPValue IN (N'Программное', N'Факт', N'Другой станок') THEN 'J' ELSE 'N' END AS [УП Готова? (J/N)],
    merged.MergedUPValue AS [Тип Тшт.],
    merged.MergedUPDate AS [Дата написания УП],

    -- Оснастка
    merged.MergedToolReady AS [Оснастка готова? (J/N)],
    merged.MergedToolDate AS [Дата готовности оснастки],
    merged.MergedToolList AS [Оснастка список],

    s.SalesQty AS [Кол-во в заказе],
    s.SalesPrice AS [Цена],
    s.Currency AS [Валюта],
    CASE WHEN s.OpenSumFC IS NULL OR s.OpenSumFC = 0 THEN ISNULL(s.SalesQty, 0) * ISNULL(s.SalesPrice, 0) ELSE s.OpenSumFC END AS [Объем (OpenSumFC)],

    merged.MergedPricePerMin AS [Цена за мин],
    CASE WHEN merged.MergedPricePerMin IS NULL THEN NULL ELSE ROUND((merged.MergedPricePerMin - 1.8) / 1.8 * 100, 2) END AS [Рентабельность %]

FROM #AllPositions b
LEFT JOIN #ChildMapping cm ON b.BELNR_ID = cm.BELNR_ID AND b.BELPOS_ID = cm.ParentPos

-- Данные родителя
LEFT JOIN #FactStart fs ON b.BELNR_ID = fs.BELNR_ID AND b.BELPOS_ID = fs.BELPOS_ID
LEFT JOIN #Material m ON b.BELNR_ID = m.BELNR_ID AND b.BELPOS_ID = m.BELPOS_ID
LEFT JOIN #UPStatus up ON b.BELNR_ID = up.BELNR_ID AND b.BELPOS_ID = up.BELPOS_ID
LEFT JOIN #ToolAgg ta ON b.BELNR_ID = ta.BELNR_ID AND b.BELPOS_ID = ta.BELPOS_ID
LEFT JOIN #Finance fin ON b.BELNR_ID = fin.BELNR_ID AND b.BELPOS_ID = fin.BELPOS_ID

-- Данные полуфабриката (21...)
LEFT JOIN #FactStart c_fs ON b.BELNR_ID = c_fs.BELNR_ID AND cm.ChildPos = c_fs.BELPOS_ID
LEFT JOIN #Material c_m ON b.BELNR_ID = c_m.BELNR_ID AND cm.ChildPos = c_m.BELPOS_ID
LEFT JOIN #UPStatus c_up ON b.BELNR_ID = c_up.BELNR_ID AND cm.ChildPos = c_up.BELPOS_ID
LEFT JOIN #ToolAgg c_ta ON b.BELNR_ID = c_ta.BELNR_ID AND cm.ChildPos = c_ta.BELPOS_ID
LEFT JOIN #Finance c_fin ON b.BELNR_ID = c_fin.BELNR_ID AND cm.ChildPos = c_fin.BELPOS_ID

-- Продажи
LEFT JOIN #SalesData s ON b.BELNR_ID = s.BELNR_ID AND b.BELPOS_ID = s.BELPOS_ID
CROSS JOIN #DateRange d

-- Схлопывание данных ребенка в строку родителя
OUTER APPLY (
    SELECT 
        MergedFactStart = CASE 
            WHEN fs.FactStartDate IS NULL THEN c_fs.FactStartDate
            WHEN c_fs.FactStartDate IS NULL THEN fs.FactStartDate
            WHEN fs.FactStartDate < c_fs.FactStartDate THEN fs.FactStartDate
            ELSE c_fs.FactStartDate
        END,
        MergedIssuedQty = NULLIF(ISNULL(m.MaterialIssuedQty, 0) + ISNULL(c_m.MaterialIssuedQty, 0), 0),
        MergedMatDesc = STUFF(
            COALESCE(', ' + m.MaterialDescription, '') + 
            COALESCE(', ' + c_m.MaterialDescription, ''), 
        1, 2, ''),
        
        -- УП: приоритет отдаем готовому значению полуфабриката, затем родителя
        MergedUPValue = CASE 
            WHEN c_up.UP_Value IN (N'Программное', N'Факт', N'Другой станок') THEN c_up.UP_Value
            WHEN up.UP_Value IN (N'Программное', N'Факт', N'Другой станок') THEN up.UP_Value
            ELSE COALESCE(c_up.UP_Value, up.UP_Value)
        END,
        MergedUPDate = CASE 
            WHEN c_up.UP_Value IN (N'Программное', N'Факт', N'Другой станок') THEN COALESCE(c_up.UP_Date, up.UP_Date)
            WHEN up.UP_Value IN (N'Программное', N'Факт', N'Другой станок') THEN COALESCE(up.UP_Date, c_up.UP_Date)
            ELSE COALESCE(c_up.UP_Date, up.UP_Date)
        END,

        MergedToolList = STUFF(
            COALESCE(', ' + ta.ToolingList, '') + 
            COALESCE(', ' + c_ta.ToolingList, ''), 
        1, 2, ''),
        MergedToolReady = CASE 
            WHEN ta.ToolingReady = 'N' OR c_ta.ToolingReady = 'N' THEN 'N'
            WHEN ta.ToolingReady = 'J' OR c_ta.ToolingReady = 'J' THEN 'J'
            ELSE NULL END,
        MergedToolDate = CASE 
            WHEN ta.ToolingReadyDate IS NULL THEN c_ta.ToolingReadyDate
            WHEN c_ta.ToolingReadyDate IS NULL THEN ta.ToolingReadyDate
            WHEN ta.ToolingReadyDate > c_ta.ToolingReadyDate THEN ta.ToolingReadyDate 
            ELSE c_ta.ToolingReadyDate END,
            
        -- Финансы: берем с проводки мехобработки полуфабриката 21...
        MergedPricePerMin = COALESCE(c_fin.PricePerMin, fin.PricePerMin)
) merged

WHERE
    b.ItemCode NOT LIKE '21%'
    AND (
        (b.LFGDAT >= d.FilterStart AND b.LFGDAT <= d.FilterEnd)
        OR (b.GanttOrderEnd >= d.FilterStart AND b.GanttOrderEnd <= d.FilterEnd)
        OR (merged.MergedFactStart >= d.FilterStart AND merged.MergedFactStart <= d.FilterEnd)
        OR (
            ISNULL(merged.MergedFactStart, b.LFGDAT) <= d.FilterEnd
            AND ISNULL(b.GanttOrderEnd, b.LFGDAT) >= d.FilterStart
        )
    )
ORDER BY b.DocEntry, b.BELNR_ID, b.BELPOS_ID;
